import { expect, it } from "vitest";
import { ValidationError, wallToUtc } from "../src/admin/validation-wall-time";

it("rejects gaps and unknown zones", () => {
  expect(() => wallToUtc("2026-03-29 01:30", "Europe/London")).toThrow(ValidationError);
  expect(() => wallToUtc("2026-10-25 01:30", "Mars/Olympus")).toThrow(ValidationError);
});

// Fold assertion
it("chooses the later occurrence of an autumn fold", () => {
  expect(wallToUtc("2026-10-25 01:30", "Europe/London").toISOString()).toBe("2026-10-25T01:30:00.000Z");
});
